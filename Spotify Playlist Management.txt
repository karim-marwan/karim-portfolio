#include <iostream>

class Song
{
private:
    int id, playCount;
    string name, artist;

public:
    Song() : id(0), playCount(0), name(""), artist("") {}

    Song(int id, string name, string artist) : id(id), name(name), artist(artist), playCount(0) {}

    void incrementPlayCount()
    {
        playCount++;
    }

    int getId() { return id; }
    string getName() { return name; }
    string getArtist() { return artist; }
    int getPlayCount() { return playCount; }

    void displaySong()
    {
        cout << "ID: "<<"(" << id << ")" << " | Name: " << "(" << name << ")" << " | Artist: " << "(" <<  artist << ")" 
            << " | Plays: " << "(" << playCount << ")" << endl;
    }
};


class PlaylistNode
{
public:
    Song data;
    PlaylistNode* next;
    PlaylistNode* prev;

    PlaylistNode(Song s) : data(s), next(nullptr), prev(nullptr) {}
};


class Playlist
{
private:
    string playlistName;
    PlaylistNode* head;
    PlaylistNode* tail;

public:
    Playlist(string name) : playlistName(name), head(nullptr), tail(nullptr) {}

    bool isEmpty()
    {
        return head == nullptr;
    }

    

    void addSong(Song s)
    {
        // Check if the song already exists in the playlist (by ID)
        PlaylistNode* temp = head;
        while (temp)
        {
            if (temp->data.getId() == s.getId())
            {
                // Song already exists, do not add it again
                cout << "Song already exists in playlist \"" << playlistName << "\"!" << endl;
                return;
            }
            temp = temp->next;
        }

        // Create a new node for the song
        PlaylistNode* newNode = new PlaylistNode(s);

        if (isEmpty())
        {
            head = tail = newNode;
        }
        else
        {
            tail->next = newNode;
            newNode->prev = tail;
            tail = newNode;
        }

        cout << "Song added successfully to playlist \"" << playlistName << "\"!" << endl;
    }


    bool deleteSong(int songId)
    {
        PlaylistNode* temp = head;

        while (temp && temp->data.getId() != songId)
        {
            temp = temp->next;
        }

        if (!temp)
        {
            cout << "Song doesn't exist!" << endl;
            return false;
        }

        if (head == tail)
        {
            head = tail = nullptr;
        }
        else
        {
            if (temp->prev)
                temp->prev->next = temp->next;
            else
                head = temp->next;

            if (temp->next)
                temp->next->prev = temp->prev;
            else
                tail = temp->prev;
        }

        delete temp;
        cout << "Song deleted successfully from playlist \"" << playlistName << "\"!" << endl;
        return true;
    }

    void displayPlaylist()
    {
        if (isEmpty())
        {
            cout << "Playlist \"" << playlistName << "\" is empty!" << endl;
            return;
        }

        cout << "\nPlaylist: " << playlistName << endl;

        PlaylistNode* temp = head;
        while (temp)
        {
            temp->data.displaySong();
            temp = temp->next;
        }
    }

    int countSongs()
    {
        int count = 0;
        PlaylistNode* temp = head;

        while (temp)
        {
            count++;
            temp = temp->next;
        }

        return count;
    }

    PlaylistNode* getHead()
    {
        return head;
    }

    void sortByMostPlays()
    {
        if (!head || !head->next) return;

        PlaylistNode* i = head;

        while (i)
        {
            PlaylistNode* j = i->next;

            while (j)
            {
                if (i->data.getPlayCount() < j->data.getPlayCount())
                {
                    Song temp = i->data;
                    i->data = j->data;
                    j->data = temp;
                }

                j = j->next;
            }

            i = i->next;
        }

        cout << "Playlist sorted by play count (most played first)." << endl;
    }

    void sortByLeastPlays()
    {
        if (!head || !head->next) return;

        PlaylistNode* i = head;

        while (i)
        {
            PlaylistNode* j = i->next;

            while (j)
            {
                if (i->data.getPlayCount() > j->data.getPlayCount())
                {
                    Song temp = i->data;
                    i->data = j->data;
                    j->data = temp;
                }

                j = j->next;
            }

            i = i->next;
        }

        cout << "Playlist sorted by play count (least played first)." << endl;
    }
};



class LibraryNode
{
public:
    Song data;
    LibraryNode* next;

    LibraryNode(Song s) : data(s), next(nullptr) {}
};

class MusicLibrary
{
private:
    LibraryNode* head;
    LibraryNode* tail;

public:
    MusicLibrary() : head(nullptr), tail(nullptr) {}

    void addSong(Song s)
    {
        LibraryNode* temp = new LibraryNode(s);

        if (!head)
        {
            head = tail = temp;
        }
        else
        {
            tail->next = temp;
            tail = temp;
        }
    }

    void displaySongs()
    {
        if (!head)
        {
            cout << "Library empty . . .!" << endl;
            return;
        }

        cout << "\n* * * Music Library * * * " << endl;

        LibraryNode* temp = head;
        while (temp)
        {
            temp->data.displaySong();
            temp = temp->next;
        }
    }

    Song* searchByName(const string& name)
    {
        LibraryNode* temp = head;

        while (temp)
        {
            if (temp->data.getName() == name)
                return &temp->data;

            temp = temp->next;
        }

        return nullptr;
    }

    Song* searchByArtist(const string& artist)
    {
        LibraryNode* temp = head;

        while (temp)
        {
            if (temp->data.getArtist() == artist)
                return &temp->data;

            temp = temp->next;
        }

        return nullptr;
    }

    Song* searchById(int id)
    {
        LibraryNode* temp = head;

        while (temp)
        {
            if (temp->data.getId() == id)
                return &temp->data;

            temp = temp->next;
        }

        return nullptr;
    }
};



class Player
{
private:
    Song* lastPlayed;
    queue<Song*> recentlyPlayed;
    const int maxRecent = 7; // Max history size

public:
    Player() : lastPlayed(nullptr) {}

    void playSong(Song* s) // Play song and update play count and history
    {
        if (!s)
        {
            cout << "Cannot play: Song not found!" << endl;
            return;
        }

        s->incrementPlayCount();
        lastPlayed = s;

        recentlyPlayed.push(s);

        if (recentlyPlayed.size() > maxRecent)
        {
            recentlyPlayed.pop();
        }

        cout << "\nNow playing:" << endl;
        s->displaySong();
    }

    void showLastPlayed()
    {
        if (!lastPlayed)
        {
            cout << "No song has been played yet." << endl;
        }
        else
        {
            cout << "\nLast Played Song:" << endl;
            lastPlayed->displaySong();
        }
    }

    void showRecentlyPlayed()
    {
        if (recentlyPlayed.empty())
        {
            cout << "No recently played songs." << endl;
            return;
        }

        cout << "\nRecently Played Songs:" << endl;

        queue<Song*> tempQueue = recentlyPlayed;

        while (!tempQueue.empty())
        {
            tempQueue.front()->displaySong();
            tempQueue.pop();
        }
    }
};


class SearchHistory
{
private:
    stack<string> history;

public:
    void addSearch(const string& s)
    {
        history.push(s);
    }

    void displayHistory()
    {
        if (history.empty())
        {
            cout << "No search history . . . !" << endl;
            return;
        }

        cout << "\nSearch History (Most recent first):" << endl;

        stack<string> tempStack = history;

        while (!tempStack.empty())
        {
            cout << "- " << tempStack.top() << endl;
            tempStack.pop();
        }
    }
};


class ShuffleNode
{
public:
    Song data;
    ShuffleNode* left;
    ShuffleNode* right;

    ShuffleNode(Song s) : data(s), left(nullptr), right(nullptr) {}
};

class ShuffleBST
{
private:
    ShuffleNode* root;

    ShuffleNode* insert(ShuffleNode* temp, Song s)
    {
        if (!temp)
            return new ShuffleNode(s);

        if (s.getName() < temp->data.getName())
            temp->left = insert(temp->left, s);
        else
            temp->right = insert(temp->right, s);

        return temp;
    }

    void inorder(ShuffleNode* temp)
    {
        if (!temp) return;
        inorder(temp->left);
        temp->data.displaySong();
        inorder(temp->right);
    }

    void preorder(ShuffleNode* temp)
    {
        if (!temp) return;
        temp->data.displaySong();
        preorder(temp->left);
        preorder(temp->right);
    }

    void postorder(ShuffleNode* temp)
    {
        if (!temp) return;
        postorder(temp->left);
        postorder(temp->right);
        temp->data.displaySong();
    }

public:
    ShuffleBST() : root(nullptr) {}

    void buildFromPlaylist(PlaylistNode* head)
    {
        root = nullptr;

        PlaylistNode* temp = head;

        while (temp)
        {
            root = insert(root, temp->data);
            temp = temp->next;
        }
    }

    void shuffle(int type)
    {
        if (type == 1)
            inorder(root);
        else if (type == 2)
            preorder(root);
        else if (type == 3)
            postorder(root);
        else
            cout << "Invalid shuffle type!" << endl;
    }
};


class MyFileHandler
{
private:
    string myPlaylistFile, myLibraryFile;

public:
    MyFileHandler(string plFile = "playlist.txt", string libFile = "musics.txt") : myPlaylistFile(plFile), myLibraryFile(libFile) {}

    void savePlaylist(Playlist& p)   // Save the current playlist to a file
    {
        ofstream writer(myPlaylistFile);

        if (!writer)
        {
            cout << "Error opening " << myPlaylistFile << endl;
            return;
        }

        PlaylistNode* temp = p.getHead();

        while (temp)
        {
            writer << temp->data.getId() << "," << temp->data.getName() << "," << temp->data.getArtist() << "\n";
            temp = temp->next;
        }

        writer.close();
    }

    void loadPlaylist(Playlist& p) // Load playlist from file
    {
        ifstream reader(myPlaylistFile); 

        if (!reader)
        {
            cout << "No existing playlist found." << endl;
            return;
        }

        string currentLine;

        while (getline(reader, currentLine))
        {
            if (currentLine.empty()) 
                continue; 

            size_t pos1 = currentLine.find(','); // Find first comma
            size_t pos2 = currentLine.rfind(',');  // Find last comma

            if (pos1 == string::npos || pos2 == string::npos || pos1 == pos2)  // If commas missing or only one comma invalid skip it
                continue;  

            int id = stoi(currentLine.substr(0, pos1));  // Extract substring before first comma and convert to int (Song ID)
            string name = currentLine.substr(pos1 + 1, pos2 - pos1 - 1);  // Extract substring between first and last comma (Song Name)
            string artist = currentLine.substr(pos2 + 1); // Extract substring after last comma  (Artist Name)

            p.addSong(Song(id, name, artist)); // Create Song object and add to Playlist
        }

        reader.close();
    }

    void removeSongFromFile(int songId)
    {
        ifstream reader(myPlaylistFile);

        if (!reader) // If file dot exist
        {
            cout << "File not found!" << endl;
            return;
        }

        string content = ""; // Will store all lines except the one to delete
        string line;   // Temporary variable for each line

        while (getline(reader, line))
        {
            if (line.empty()) continue;

            int id = stoi(line.substr(0, line.find(','))); // Extract ID from line (before first comma) and convert to int

            if (id != songId) // If this is NOT the song to remove
            {
                content += line + "\n";  // Keep it in the content
            }
        }
        reader.close();

        ofstream writer(myPlaylistFile, ios::trunc);  //erases all content from the old file before writing.
        writer << content; // Write updated content back to file
        writer.close();
    }

};


// Add selected song from library to playlist and save to file
void addSongToPlaylistFromLibrary(Playlist& playlist, MusicLibrary& library, MyFileHandler& fileHandler)
{
    library.displaySongs();

    cout << "Enter the ID of the song to add : ";
    int id;
    cin >> id;

    Song* s = library.searchById(id);

    if (s)
    {
        playlist.addSong(*s);
        fileHandler.savePlaylist(playlist);
    }
    else
    {
        cout << "Song not found in library!" << endl;
    }
}
